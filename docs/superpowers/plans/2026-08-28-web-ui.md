> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 Groot 增加内嵌 Web 界面：登录、聊天（SSE 流式）、仪表盘、设置弹窗（含能力总览），单二进制交付。

**Tech Stack:** Go + Hertz（现有）；Vue 3 + TypeScript + Vite + Element Plus + Pinia + Vue Router + markdown-it + highlight.js。

```go
// Package websession 提供 Web 界面登录会话的内存存储：
// 随机令牌 + TTL 过期，以及按客户端 IP 的登录失败限速。
// 服务重启后全部会话失效（需重新登录），符合单账号定位。
package websession

import (
	"crypto/rand"
	"encoding/hex"
	"sync"
	"time"
)

// CookieName Web 登录会话 Cookie 名称（handler 与 middleware 共用）
const CookieName = "groot_web_session"

const (
	maxFailures   = 5                // 窗口内允许的最大失败次数
	failureWindow = 10 * time.Minute // 失败计数滑动窗口
)

// Store 登录会话存储
type Store struct {
	mu       sync.Mutex
	ttl      time.Duration
	sessions map[string]time.Time   // token -> 过期时间
	failures map[string][]time.Time // ip -> 窗口内失败时间点
	now      func() time.Time       // 可注入时钟，便于测试
}

// NewStore 创建存储，ttl 为登录会话有效期
func NewStore(ttl time.Duration) *Store {
	return &Store{
		ttl:      ttl,
		sessions: make(map[string]time.Time),
		failures: make(map[string][]time.Time),
		now:      time.Now,
	}
}

// TTL 返回会话有效期
func (s *Store) TTL() time.Duration { return s.ttl }

// Create 生成并登记一个新令牌（64 位 hex）
func (s *Store) Create() string {
	b := make([]byte, 32)
	_, _ = rand.Read(b)
	token := hex.EncodeToString(b)
	s.mu.Lock()
	defer s.mu.Unlock()
	s.sessions[token] = s.now().Add(s.ttl)
	return token
}

// Validate 校验令牌有效性，过期令牌顺带清除
func (s *Store) Validate(token string) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	exp, ok := s.sessions[token]
	if !ok {
		return false
	}
	if s.now().After(exp) {
		delete(s.sessions, token)
		return false
	}
	return true
}

// Delete 注销令牌
func (s *Store) Delete(token string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.sessions, token)
}

// RecordFailure 记录一次该 IP 的登录失败
func (s *Store) RecordFailure(ip string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.failures[ip] = append(s.pruneLocked(ip), s.now())
}

// IsLocked 判断该 IP 是否因失败过多被锁定
func (s *Store) IsLocked(ip string) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	recent := s.pruneLocked(ip)
	s.failures[ip] = recent
	return len(recent) >= maxFailures
}

// ClearFailures 登录成功后清零失败计数
func (s *Store) ClearFailures(ip string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.failures, ip)
}

// pruneLocked 剔除窗口外的失败记录（调用方须持锁）
func (s *Store) pruneLocked(ip string) []time.Time {
	cutoff := s.now().Add(-failureWindow)
	var recent []time.Time
	for _, ts := range s.failures[ip] {
		if ts.After(cutoff) {
			recent = append(recent, ts)
		}
	}
	return recent
}
```

```go
// WebAuthHandler 处理 Web 界面登录相关端点
type WebAuthHandler struct {
	enabled  bool
	username string
	password string // 已展开环境变量后的明文
	store    *websession.Store
	logger   *logger.Logger
}

// NewWebAuthHandler 创建 Web 登录处理器；密码中的 ${ENV} 引用在此展开
func NewWebAuthHandler(cfg config.WebConfig, store *websession.Store, log *logger.Logger) *WebAuthHandler {
	return &WebAuthHandler{
		enabled:  cfg.Enabled,
		username: cfg.Username,
		password: config.ExpandEnv(cfg.Password),
		store:    store,
		logger:   log,
	}
}

type webLoginRequest struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

// Login 处理 POST /web/login
func (h *WebAuthHandler) Login(ctx context.Context, rc *app.RequestContext) {
	if !h.enabled {
		rc.JSON(200, map[string]interface{}{"status": "ok", "auth_required": false})
		return
	}
	ip := rc.ClientIP()
	if h.store.IsLocked(ip) {
		rc.JSON(429, map[string]interface{}{"status": "locked", "message": "登录失败次数过多，请稍后再试"})
		return
	}
	var req webLoginRequest
	if err := json.Unmarshal(rc.Request.Body(), &req); err != nil {
		rc.JSON(400, map[string]interface{}{"status": "invalid_request", "message": "请求格式错误"})
		return
	}
	userOK := subtle.ConstantTimeCompare([]byte(req.Username), []byte(h.username)) == 1
	passOK := subtle.ConstantTimeCompare([]byte(req.Password), []byte(h.password)) == 1
	// 配置密码为空视为配置错误，一律拒绝，防止空密码直通
	if h.password == "" || !userOK || !passOK {
		h.store.RecordFailure(ip)
		h.logger.Warn("Web 登录失败")
		rc.JSON(401, map[string]interface{}{"status": "unauthorized", "message": "用户名或密码错误"})
		return
	}
	h.store.ClearFailures(ip)
	token := h.store.Create()
	// secure=false：Groot 常以 http 在内网/本机运行；如经 https 反代部署，Cookie 仍随请求走 https
	rc.SetCookie(websession.CookieName, token, int(h.store.TTL().Seconds()), "/", "",
		protocol.CookieSameSiteStrictMode, false, true)
	rc.JSON(200, map[string]interface{}{"status": "ok"})
}

// Logout 处理 POST /web/logout
func (h *WebAuthHandler) Logout(ctx context.Context, rc *app.RequestContext) {
	if token := string(rc.Cookie(websession.CookieName)); token != "" {
		h.store.Delete(token)
	}
	// maxAge=-1 让浏览器立即删除 Cookie
	rc.SetCookie(websession.CookieName, "", -1, "/", "",
		protocol.CookieSameSiteStrictMode, false, true)
	rc.JSON(200, map[string]interface{}{"status": "ok"})
}

// Me 处理 GET /web/me，前端据此判断是否需要跳登录页
func (h *WebAuthHandler) Me(ctx context.Context, rc *app.RequestContext) {
	authenticated := true
	if h.enabled {
		token := string(rc.Cookie(websession.CookieName))
		authenticated = token != "" && h.store.Validate(token)
	}
	rc.JSON(200, map[string]interface{}{
		"authenticated": authenticated,
		"auth_required": h.enabled,
	})
}
```

```go
import (
	"context"
	"io/fs"
	"mime"
	"path/filepath"
	"strings"

	"github.com/cloudwego/hertz/pkg/app/server"

const webUIPlaceholder = `<!DOCTYPE html>
<html lang="zh"><head><meta charset="utf-8"><title>Groot</title></head>
<body style="font-family:sans-serif;padding:40px">
<h2>Groot Web UI 未构建</h2>
<p>请在项目根目录执行 <code>make web</code> 后重新编译 groot。</p>
</body></html>`

// NewWebUIHandler 返回服务 /ui/* 的处理函数。
// fsys 以 dist 为根；未命中的路径回退 index.html（支持前端 history 路由），
// 连 index.html 都不存在时返回未构建提示页。
func NewWebUIHandler(fsys fs.FS) app.HandlerFunc {
	return func(ctx context.Context, rc *app.RequestContext) {
		p := strings.TrimPrefix(string(rc.URI().Path()), "/ui")
		p = strings.TrimPrefix(p, "/")
		if p == "" {
			p = "index.html"
		}
		data, err := fs.ReadFile(fsys, p)
		if err != nil {
			p = "index.html"
			data, err = fs.ReadFile(fsys, p)
			if err != nil {
				rc.SetContentType("text/html; charset=utf-8")
				rc.SetStatusCode(200)
				rc.WriteString(webUIPlaceholder)
				return
			}
		}
		ct := mime.TypeByExtension(filepath.Ext(p))
		if ct == "" {
			ct = "application/octet-stream"
		}
		rc.SetContentType(ct)
		rc.SetStatusCode(200)
		rc.Write(data)
	}
}

// RegisterWebUI 在 Hertz 上注册 Web 界面静态路由
func RegisterWebUI(h *server.Hertz) {
	dist, err := fs.Sub(web.DistFS, "dist")
	if err != nil {
		return // embed 缺失时不注册（理论上不会发生）
	}
	handler := NewWebUIHandler(dist)
	h.GET("/ui", handler)
	h.GET("/ui/*filepath", handler)
}
```

```ts
import { defineStore } from 'pinia'
import { ref, computed, watchEffect } from 'vue'

export type ThemeMode = 'light' | 'dark' | 'auto'

const STORAGE_KEY = 'groot-theme'

export const useThemeStore = defineStore('theme', () => {
  const saved = (localStorage.getItem(STORAGE_KEY) as ThemeMode | null) || 'auto'
  const mode = ref<ThemeMode>(saved)

  // 系统偏好（跟随系统模式下据此决定明暗）
  const prefersDark = ref(
    window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches
  )
  if (window.matchMedia) {
    window
      .matchMedia('(prefers-color-scheme: dark)')
      .addEventListener('change', (e) => {
        prefersDark.value = e.matches
      })
  }

  const isDark = computed(() => {
    if (mode.value === 'auto') return prefersDark.value
    return mode.value === 'dark'
  })

  // 将明暗状态同步到 <html class="dark">，供 Element Plus 暗色变量生效。
  watchEffect(() => {
    const el = document.documentElement
    if (isDark.value) el.classList.add('dark')
    else el.classList.remove('dark')
  })

  function setMode(next: ThemeMode) {
    mode.value = next
    localStorage.setItem(STORAGE_KEY, next)
  }

  return { mode, isDark, setMode }
})
```

`web/src/api/sse.ts`：

```ts
import type { Attachment, SseEvent } from './types'

export interface StreamHandle {
  sessionId: string
  chatId: string
}

export interface StreamChatOptions {
  instruction: string
  prompt?: string
  attachments?: Attachment[]
  sessionId?: string
  model?: string
  agentName?: string
  signal: AbortSignal
  onStart?: (h: StreamHandle) => void
  onEvent: (e: SseEvent) => void
}

// POST /chat 并逐行解析 SSE（协议：data: {json} 行，data: [DONE] 结束）。
// 取消 = signal abort 断开连接，服务端随连接断开取消执行。
export async function streamChat(opts: StreamChatOptions): Promise<void> {
  const headers: Record<string, string> = { 'Content-Type': 'application/json' }
  if (opts.sessionId) headers['X-Session-ID'] = opts.sessionId
  if (opts.model) headers['X-Model-Name'] = opts.model
  if (opts.agentName) headers['X-Agent-Name'] = opts.agentName

  const res = await fetch('/chat', {
    method: 'POST',
    headers,
    credentials: 'same-origin',
    signal: opts.signal,
    body: JSON.stringify({
      instruction: opts.instruction,
      prompt: opts.prompt || undefined,
      attachments: opts.attachments?.length ? opts.attachments : undefined,
    }),
  })
  if (res.status === 401) throw new Error('unauthorized')
  if (res.status === 409) throw new Error('chat_limit_exceeded')
  if (!res.ok || !res.body) throw new Error(`HTTP ${res.status}`)

  opts.onStart?.({
    sessionId: res.headers.get('X-Session-ID') || '',
    chatId: res.headers.get('X-Chat-ID') || '',
  })

  const reader = res.body.getReader()
  const decoder = new TextDecoder()
  let buf = ''
  for (;;) {
    const { done, value } = await reader.read()
    if (done) break
    buf += decoder.decode(value, { stream: true })
    let idx: number
    while ((idx = buf.indexOf('\n')) >= 0) {
      const line = buf.slice(0, idx).trim()
      buf = buf.slice(idx + 1)
      if (!line.startsWith('data:')) continue
      const payload = line.slice(5).trim()
      if (payload === '[DONE]') return
      try {
        opts.onEvent(JSON.parse(payload) as SseEvent)
      } catch {
        /* 忽略无法解析的行 */
      }
    }
  }
}
```

`web/src/stores/chat.ts`：

```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { api } from '../api/client'
import { streamChat } from '../api/sse'
import type {
  Attachment, ChatRecord, SessionDetailResponse, SessionInfo, SessionListResponse, SseEvent,
} from '../api/types'

export interface ToolCallView {
  id: string
  name: string
  arguments: string
  result?: string
  isError?: boolean
  done: boolean
}

export interface AssistantTurn {
  kind: 'assistant'
  agentName?: string
  reasoning: string
  content: string
  toolCalls: ToolCallView[]
  finishReason?: string
}

export interface UserTurn {
  kind: 'user'
  content: string
  attachmentNames: string[]
}

export type Turn = UserTurn | AssistantTurn

export const useChatStore = defineStore('chat', () => {
  const sessionId = ref('')
  const turns = ref<Turn[]>([])
  const streaming = ref(false)
  const errorMsg = ref('')
  const stats = ref<ChatRecord | null>(null)
  const sessions = ref<SessionInfo[]>([])
  const sessionsTotal = ref(0)
  let controller: AbortController | null = null

  function currentAssistant(agentName?: string): AssistantTurn {
    const last = turns.value[turns.value.length - 1]
    // 子 Agent 事件切换 agent 名时另起一个回复块，标注来源
    if (last?.kind === 'assistant' && (last.agentName || '') === (agentName || '')) return last
    const t: AssistantTurn = { kind: 'assistant', agentName, reasoning: '', content: '', toolCalls: [] }
    turns.value.push(t)
    return t
  }

  function applyEvent(e: SseEvent) {
    if (e.event === 'error') {
      errorMsg.value = e.message || '执行出错'
      return
    }
    if (e.role === 'tool') {
      // 工具结果：按 tool_call_id 回填
      for (const t of turns.value) {
        if (t.kind !== 'assistant') continue
        const tc = t.toolCalls.find((c) => c.id === e.tool_call_id)
        if (tc) {
          tc.result = e.content || ''
          tc.isError = !!e.error
          tc.done = true
          return
        }
      }
      return
    }
    const turn = currentAssistant(e.agent_name)
    if (e.reasoning_content) turn.reasoning += e.reasoning_content
    if (e.content) turn.content += e.content
    if (e.tool_calls) {
      for (const c of e.tool_calls) {
        turn.toolCalls.push({ id: c.id, name: c.function.name, arguments: c.function.arguments, done: false })
      }
    }
    if (e.finish_reason) turn.finishReason = e.finish_reason
  }

  async function send(instruction: string, model: string, agentName: string, attachments: Attachment[]) {
    if (streaming.value) return
    errorMsg.value = ''
    stats.value = null
    turns.value.push({ kind: 'user', content: instruction, attachmentNames: attachments.map((a) => a.name) })
    streaming.value = true
    controller = new AbortController()
    try {
      await streamChat({
        instruction,
        attachments,
        sessionId: sessionId.value || undefined,
        model: model || undefined,
        agentName: agentName || undefined,
        signal: controller.signal,
        onStart: (h) => {
          if (h.sessionId) sessionId.value = h.sessionId
        },
        onEvent: applyEvent,
      })
      await fetchStats()
    } catch (e) {
      const msg = String(e)
      if (msg.includes('AbortError') || (e as Error).name === 'AbortError') {
        errorMsg.value = '已停止生成'
      } else if (msg.includes('chat_limit_exceeded')) {
        errorMsg.value = '该会话有正在进行的对话，请稍后再试'
      } else if (!msg.includes('unauthorized')) {
        errorMsg.value = '连接中断：' + msg
      }
    } finally {
      streaming.value = false
      controller = null
      loadSessions()
    }
  }

  function stop() {
    controller?.abort()
  }

  async function fetchStats() {
    if (!sessionId.value) return
    try {
      stats.value = await api<ChatRecord>(`/chat/${sessionId.value}`)
    } catch {
      /* 统计失败不影响对话 */
    }
  }

  async function loadSessions(limit = 50, offset = 0) {
    try {
      const r = await api<SessionListResponse>(`/sess/history?limit=${limit}&offset=${offset}`)
      sessions.value = r.sessions
      sessionsTotal.value = r.total
    } catch {
      /* 列表失败静默 */
    }
  }

  async function openSession(sid: string) {
    const r = await api<SessionDetailResponse>(`/sess/${sid}`)
    sessionId.value = sid
    errorMsg.value = ''
    stats.value = null
    turns.value = []
    for (const m of r.history.messages) {
      turns.value.push({ kind: 'user', content: m.instruction, attachmentNames: [] })
      turns.value.push({
        kind: 'assistant', agentName: m.agent_name, reasoning: '', content: m.result, toolCalls: [],
      })
    }
    fetchStats()
  }

  function newSession() {
    sessionId.value = ''
    turns.value = []
    errorMsg.value = ''
    stats.value = null
  }

  return {
    sessionId, turns, streaming, errorMsg, stats, sessions, sessionsTotal,
    send, stop, loadSessions, openSession, newSession,
  }
})
```

```vue
<script setup lang="ts">
import { onMounted, ref } from 'vue'
import { useChatStore } from '../../stores/chat'
import { useMetaStore } from '../../stores/meta'
import type { Attachment } from '../../api/types'

const chat = useChatStore()
const meta = useMetaStore()
const text = ref('')
const model = ref('')
const attachments = ref<Attachment[]>([])
const fileInput = ref<HTMLInputElement | null>(null)

onMounted(async () => {
  await meta.loadModels()
  model.value = meta.models?.default || ''
})

function pickFile() {
  fileInput.value?.click()
}

function typeByName(name: string): string {
  const ext = name.split('.').pop()?.toLowerCase() || ''
  if (['png', 'jpg', 'jpeg'].includes(ext)) return 'image'
  return 'file'
}

async function onFiles(e: Event) {
  const files = (e.target as HTMLInputElement).files
  if (!files) return
  for (const f of Array.from(files)) {
    const base64 = await new Promise<string>((resolve) => {
      const r = new FileReader()
      r.onload = () => resolve((r.result as string).split(',')[1] || '')
      r.readAsDataURL(f)
    })
    attachments.value.push({ type: typeByName(f.name), name: f.name, content: base64 })
  }
  ;(e.target as HTMLInputElement).value = ''
}

function submit() {
  const instruction = text.value.trim()
  if (!instruction || chat.streaming) return
  chat.send(instruction, model.value, '', attachments.value)
  text.value = ''
  attachments.value = []
}

function onKeydown(e: KeyboardEvent) {
  if (e.key === 'Enter' && !e.shiftKey) {
    e.preventDefault()
    submit()
  }
}
</script>

<template>
  <div class="input-card">
    <div v-if="attachments.length" class="att-row">
      <el-tag v-for="(a, i) in attachments" :key="i" closable size="small" @close="attachments.splice(i, 1)">
```

```python
"""Web 登录端点系统测试。

运行前置：groot 服务已启动。
- security.web 关闭时：跳过登录类用例，仅验证 /web/me 与 /ui。
- security.web 开启时：需设置环境变量 GROOT_WEB_USER / GROOT_WEB_PASS 为配置的账号。
"""
import os

import pytest
import requests

BASE_URL = os.environ.get("GROOT_BASE_URL", "http://localhost:8080")
WEB_USER = os.environ.get("GROOT_WEB_USER", "admin")
WEB_PASS = os.environ.get("GROOT_WEB_PASS", "")


def _me():
    r = requests.get(f"{BASE_URL}/web/me", timeout=5)
    r.raise_for_status()
    return r.json()


def _auth_enabled():
    return _me().get("auth_required", False)


def test_me_endpoint():
    """/web/me 返回 authenticated 与 auth_required 字段"""
    data = _me()
    assert "authenticated" in data
    assert "auth_required" in data


def test_ui_served():
    """/ui/ 返回 HTML 页面（构建后为 SPA，未构建为提示页）"""
    r = requests.get(f"{BASE_URL}/ui/", timeout=5)
    assert r.status_code == 200
    assert "html" in r.headers.get("Content-Type", "")


@pytest.mark.skipif(not _auth_enabled(), reason="security.web 未开启")
def test_login_wrong_password():
    """错误密码返回 401"""
    r = requests.post(
        f"{BASE_URL}/web/login",
        json={"username": WEB_USER, "password": "definitely-wrong"},
        timeout=5,
    )
    assert r.status_code == 401


@pytest.mark.skipif(not _auth_enabled(), reason="security.web 未开启")
def test_login_and_cookie_access():
    """正确登录后，携带 Cookie 可访问受保护端点；登出后失效"""
    s = requests.Session()
    r = s.post(
        f"{BASE_URL}/web/login",
        json={"username": WEB_USER, "password": WEB_PASS},
        timeout=5,
    )
    assert r.status_code == 200, r.text
    assert "groot_web_session" in s.cookies

    r = s.get(f"{BASE_URL}/sess/history", timeout=5)
    assert r.status_code == 200

    r = s.get(f"{BASE_URL}/web/me", timeout=5)
    assert r.json()["authenticated"] is True

    r = s.post(f"{BASE_URL}/web/logout", timeout=5)
    assert r.status_code == 200


@pytest.mark.skipif(not _auth_enabled(), reason="security.web 未开启")
def test_unauthenticated_api_rejected():
    """无凭证访问受保护端点返回 401（API 认证开启场景）"""
    r = requests.get(f"{BASE_URL}/sess/history", timeout=5)
    # 若 API Key 认证也未开启，此请求会 200——两种认证均开启时才断言 401
    if r.status_code == 200:
        pytest.skip("API 认证未开启，跳过")
    assert r.status_code == 401
```
