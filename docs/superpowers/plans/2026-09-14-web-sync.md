# Web 工作空间配置同步 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Web 工作空间面板中提供本地 `~/.groot/` 与数据库之间的配置差异查看与双向同步，数据库侧删除改为记录状态标记以区分「他人已删除」与「远端从无记录」。

**Architecture:** 数据库 `shared_resources` 表增加 `status` 列承载软删除；`ResourceRepo` 读方法默认过滤已删除记录，新增 `ListWithDeleted` 供差异比较使用；`ComputeDiff` 据此产出以本地为基准的 A/M/D 清单并附带远端记录元信息；Web 层新增 `SyncHandler` 直接复用既有 `sync.SyncManager`，不实现独立同步算法；前端在工作空间面板提供差异对话框，方向由用户点击决定。

**Tech Stack:** Go 1.x + sqlx（SQLite/MySQL/PostgreSQL 三方言）、Hertz、Vue 3 + Element Plus + TypeScript。测试用 Go 标准 `testing`。

**Spec:** `docs/superpowers/specs/2026-09-14-web-sync-design.md`

```go
// addMissingColumns 为已存在的表补充新版本引入的列。现有迁移是一组幂等建表语句，
// 没有版本号机制，因此老库不会因为 DDL 变更而拿到新列，需要在此显式补齐。
func addMissingColumns(db *sqlx.DB, dialect Dialect) error {
	return addColumnIfMissing(db, dialect, "shared_resources", "status", statusColumnType(dialect))
}

func statusColumnType(d Dialect) string {
	switch d {
	case DialectMySQL:
		return "VARCHAR(16) NOT NULL DEFAULT 'active'"
	case DialectPostgres:
		return "VARCHAR(16) NOT NULL DEFAULT 'active'"
	default: // SQLite
		return "TEXT NOT NULL DEFAULT 'active'"
	}
}

// addColumnIfMissing adds a column only if the catalog reports it absent.
// Probing first avoids relying on `ADD COLUMN IF NOT EXISTS` syntax variants.
func addColumnIfMissing(db *sqlx.DB, dialect Dialect, table, column, columnDef string) error {
	exists, err := columnExists(db, dialect, table, column)
	if err != nil {
		return fmt.Errorf("probe column %s.%s: %w", table, column, err)
	}
	if exists {
		return nil
	}
	stmt := fmt.Sprintf("ALTER TABLE %s ADD COLUMN %s %s", table, column, columnDef)
	if _, err := db.Exec(stmt); err != nil {
		return fmt.Errorf("add column %s.%s: %w", table, column, err)
	}
	return nil
}

// columnExists queries the dialect-specific catalog for a column by name.
// Mirrors indexExists: scope MySQL to DATABASE() and Postgres to current_schema()
// so a same-named table in another schema cannot produce a false positive.
func columnExists(db *sqlx.DB, dialect Dialect, table, column string) (bool, error) {
	var q string
	var args []interface{}
	switch dialect {
	case DialectMySQL:
		q = `SELECT COUNT(*) FROM information_schema.columns
		     WHERE table_schema = DATABASE() AND table_name = ? AND column_name = ?`
		args = []interface{}{table, column}
	case DialectPostgres:
		q = `SELECT COUNT(*) FROM information_schema.columns
		     WHERE table_schema = current_schema() AND table_name = $1 AND column_name = $2`
		args = []interface{}{table, column}
	default: // SQLite
		q = `SELECT COUNT(*) FROM pragma_table_info(?) WHERE name=?`
		args = []interface{}{table, column}
	}
	var n int
	if err := db.Get(&n, q, args...); err != nil {
		return false, err
	}
	return n > 0, nil
}
```

写法与 [migrate.go:62](internal/db/migrate.go#L62) 的 `indexExists` 保持一致：Postgres 直接写 `$1`/`$2` 而不走 `Rebind`，MySQL/SQLite 用 `?`。

		}

		remoteMap := make(map[string]*repo.ResourceEntry, len(remoteEntries))
		for _, e := range remoteEntries {
			remoteMap[e.Path] = e
			result.Remote[e.Path] = RemoteMeta{
				Deleted:   e.Status == repo.ResourceStatusDeleted,
				UpdatedAt: e.UpdatedAt,
			}
		}

		for relPath, localInfo := range localFiles {
			remote, ok := remoteMap[relPath]
			switch {
			case !ok || remote.Status == repo.ResourceStatusDeleted:
				// 远端无记录，或记录已删除 → 本地独有
				result.Added = append(result.Added, relPath)
			case localInfo.size != remote.Size || localInfo.hash != remote.ContentHash:
				result.Modified = append(result.Modified, relPath)
			default:
				result.Same = append(result.Same, relPath)
			}
			delete(remoteMap, relPath)
		}
		for relPath, remote := range remoteMap {
			// 已删除记录且本地也不存在 → 双方一致，不是差异
			if remote.Status == repo.ResourceStatusDeleted {
				continue
			}
			result.Removed = append(result.Removed, relPath)
		}
	}

// pruneEmptyDirs 从 dir 起向上逐级删除变空的目录，停在白名单根目录之上一层
// （即白名单根目录自身与 homeDir 永不删除）。目录非空或删除失败即停止，
// 不返回错误：清理失败不应让 pull 整体失败。
func pruneEmptyDirs(homeDir, dir string) {
	for {
		rel, err := filepath.Rel(homeDir, dir)
		if err != nil {
			return
		}
		rel = filepath.ToSlash(rel)
		// 到达 HOME 本身或白名单根目录（单段路径，如 "skills"）就停
		if rel == "." || rel == ".." || !strings.Contains(rel, "/") {
			return
		}
		entries, err := os.ReadDir(dir)
		if err != nil || len(entries) > 0 {
			return
		}
		if err := os.Remove(dir); err != nil {
			return
		}
		dir = filepath.Dir(dir)
	}
}

//   - "D" 远端有、本地无
type WebDiffEntry struct {
	Path            string `json:"path"`
	Status          string `json:"status"`
	RemoteDeleted   bool   `json:"remoteDeleted"`
	NeedsRestart    bool   `json:"needsRestart"`
	RemoteUpdatedAt int64  `json:"remoteUpdatedAt,omitempty"` // 毫秒；远端无记录时为 0
}

// WebDiffView 是一次差异计算的完整结果。
type WebDiffView struct {
	InSync       bool           `json:"inSync"`
	NeedsRestart bool           `json:"needsRestart"`
	Entries      []WebDiffEntry `json:"entries"`
}

// BuildWebDiff 把 DiffResult 折成按路径升序的扁平清单。
// 一致的文件（DiffResult.Same）不进入结果——面板只展示需要决策的条目。
func BuildWebDiff(d DiffResult) WebDiffView {
	view := WebDiffView{
		InSync:  d.IsEmpty(),
		Entries: []WebDiffEntry{},
	}
	add := func(paths []string, status string) {
		for _, p := range paths {
			e := WebDiffEntry{
				Path:         p,
				Status:       status,
				NeedsRestart: pathNeedsRestart(p),
			}
			if meta, ok := d.Remote[p]; ok {
				e.RemoteDeleted = meta.Deleted
				if !meta.UpdatedAt.IsZero() {
					e.RemoteUpdatedAt = meta.UpdatedAt.UnixMilli()
				}
			}
			if e.NeedsRestart {
				view.NeedsRestart = true
			}
			view.Entries = append(view.Entries, e)
		}
	}
	add(d.Added, "A")
	add(d.Modified, "M")
	add(d.Removed, "D")

	sort.Slice(view.Entries, func(i, j int) bool {
		return view.Entries[i].Path < view.Entries[j].Path
	})
	return view
}

// pathNeedsRestart 判断该路径变更后是否需要重启服务才生效。
// 复用 CLI 渲染使用的同一份前缀清单，避免两处规则漂移。
func pathNeedsRestart(rel string) bool {
	for _, prefix := range needsRestartPaths {
		if rel == prefix || strings.HasPrefix(rel, prefix) {
			return true
		}
	}
	return false
}
```

`RemoteUpdatedAt` 用 `UnixMilli()` 从 `RemoteMeta.UpdatedAt`（`time.Time`）取值，本文件自身不需要 import `time`。

package handler

import (
	"context"
	"errors"
	"strings"

	"github.com/cloudwego/hertz/pkg/app"
	"github.com/cloudwego/hertz/pkg/common/utils"

	"github.com/zfd81/groot/internal/repo"
	grootsync "github.com/zfd81/groot/internal/sync"
)

// SyncHandler 配置同步的 HTTP 绑定层；同步语义在 internal/sync。
type SyncHandler struct {
	mgr grootsync.SyncManager
}

// NewSyncHandler 构造 SyncHandler。resources 为 nil（SQLite 单机模式）时
// SyncManager 的所有方法返回 ErrSyncDisabled，端点统一回 409。
func NewSyncHandler(homeDir string, resources repo.ResourceRepo) *SyncHandler {
	return &SyncHandler{mgr: grootsync.NewSyncManager(homeDir, resources)}
}

// syncRequest 是三个端点共用的请求体。paths 省略或为空表示全量同步。
type syncRequest struct {
	Paths []string `json:"paths"`
}

// normalizeSyncPaths 去掉空白项并 trim；结果为空表示全量（交给 SyncManager 展开白名单）。
func normalizeSyncPaths(in []string) []string {
	out := make([]string, 0, len(in))
	for _, p := range in {
		if p = strings.TrimSpace(p); p != "" {
			out = append(out, p)
		}
	}
	return out
}

// syncErrorStatus 把同步错误映射为 HTTP 状态码与业务 status 值。
// 路径校验错误来自 ValidateSyncPath，它返回的是普通 error，
// 因此以「非 ErrSyncDisabled 的 sync: 前缀错误」判定为参数问题。
func syncErrorStatus(err error) (int, string) {
	switch {
	case errors.Is(err, grootsync.ErrSyncDisabled):
		return 409, "sync_disabled"
	case strings.HasPrefix(err.Error(), "sync: "):
		return 400, "invalid_request"
	default:
		return 500, "error"
	}
}

func writeSyncError(rc *app.RequestContext, err error) {
	code, status := syncErrorStatus(err)
	message := "内部错误"
	switch status {
	case "sync_disabled":
		message = "配置同步仅在 MySQL/PostgreSQL 模式下可用"
	case "invalid_request":
		message = err.Error()
	}
	rc.JSON(code, utils.H{"status": status, "message": message})
}

// bindPaths 解析请求体中的 paths；请求体缺失或非法 JSON 视为全量同步。
func bindPaths(rc *app.RequestContext) []string {
	var req syncRequest
	if err := rc.BindJSON(&req); err != nil {
		return nil
	}
	return normalizeSyncPaths(req.Paths)
}

// Diff 处理 POST /web/sync/diff —— 只读比较，不修改任何一侧。
// 用 POST 而非 GET 是为了在请求体里传路径数组。
func (h *SyncHandler) Diff(ctx context.Context, rc *app.RequestContext) {
	paths := bindPaths(rc)
	// 清理上次中断留下的 *.tmp 残留，确保差异反映真实状态
	_ = h.mgr.CleanTmpResidue(paths)

	d, err := h.mgr.Diff(paths)
	if err != nil {
		writeSyncError(rc, err)
		return
	}
	view := grootsync.BuildWebDiff(d)
	rc.JSON(200, utils.H{
		"status":       "success",
		"inSync":       view.InSync,
		"needsRestart": view.NeedsRestart,
		"entries":      view.Entries,
	})
}

// Push 处理 POST /web/sync/push —— 本地 HOME 覆盖数据库。
// 不接受前端传来的差异快照：SyncManager.Push 内部会重新计算差异后执行，
// 因此用户确认时以那一刻的真实差异为准，服务端无需保存状态。
func (h *SyncHandler) Push(ctx context.Context, rc *app.RequestContext) {
	h.apply(rc, h.mgr.Push)
}

// Pull 处理 POST /web/sync/pull —— 数据库覆盖本地 HOME。
func (h *SyncHandler) Pull(ctx context.Context, rc *app.RequestContext) {
	h.apply(rc, h.mgr.Pull)
}

func (h *SyncHandler) apply(rc *app.RequestContext, op func([]string) error) {
	paths := bindPaths(rc)
	if err := op(paths); err != nil {
		writeSyncError(rc, err)
		return
	}
	rc.JSON(200, utils.H{"status": "success"})
}
```
